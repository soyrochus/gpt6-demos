const root='/home/iwk/src/gpt6-demos/edificio-europa';
const result=await Bun.build({entrypoints:[root+'/frontend.ts'],target:'browser',minify:true,plugins:[{name:'probe-no-photographs',setup(build){build.onLoad({filter:/\/frontend\.ts$/},async({path})=>({contents:(await Bun.file(path).text()).replace(/<img\b[^>]*>/g,''),loader:'ts'}));}}]});
if(!result.success)throw new AggregateError(result.logs);
const js=(await result.outputs[0]!.text()).replaceAll('</script','<\\/script');
const css=(await Bun.file(root+'/style.css').text()).replace(/@import\s+url\([^;]+;/g,'');
const html=(await Bun.file(root+'/index.html').text()).replace('<link rel="stylesheet" href="./style.css">',()=>`<style>${css}</style>`).replace('<script type="module" src="./frontend.ts"></script>',()=>`<script>window.addEventListener('error',e=>window.ipc.postMessage(JSON.stringify({error:e.message})));</script><script type="module">${js}</script>`);
await Bun.write('/tmp/europa-native-investigation/europa-page.html',html);
console.log('Bundled Europa without photographs or remote font requests',html.length);
